/* ---------- 页面里：注册与更新提示 ---------- */
if ('serviceWorker' in navigator) {                 // ① 一定要特性检测
  window.addEventListener('load', async () => {
    try {
      const reg = await navigator.serviceWorker.register('/sw.js', { scope: '/' });
      reg.addEventListener('updatefound', () => {
        const sw = reg.installing;
        sw.addEventListener('statechange', () => {
          // ② 新版本装好了，而且当前页面已被旧 SW 控制 → 提示用户刷新
          if (sw.state === 'installed' && navigator.serviceWorker.controller) {
            showUpdateBar(() => sw.postMessage({ type: 'SKIP_WAITING' }));
          }
        });
      });
      // ③ 控制器换了 → 刷新一次，彻底切到新版本
      navigator.serviceWorker.addEventListener('controllerchange', () => location.reload());
    } catch (err) {
      console.warn('SW 注册失败（file:// 或非 HTTPS 下是正常的）：', err.name, err.message);
    }
  });
}

/* ---------- sw.js：预缓存 / 清理 / 拦截 ---------- */
const STATIC  = 'static-v3';
const OFFLINE = 'offline-v1';
const KEEP    = [STATIC, OFFLINE];                  // 白名单：别把离线兜底页删了

self.addEventListener('install', (e) => {
  e.waitUntil(caches.open(STATIC).then((c) => c.addAll(['/', '/index.html', '/style.css', '/app.js'])));
});

self.addEventListener('activate', (e) => {
  e.waitUntil((async () => {
    for (const n of await caches.keys()) if (!KEEP.includes(n)) await caches.delete(n);
    await self.clients.claim();
  })());
});

self.addEventListener('fetch', (e) => {
  e.respondWith(handleFetch(e.request));            // 决策器挂在这里
});

self.addEventListener('message', (e) => {
  if (e.data && e.data.type === 'SKIP_WAITING') self.skipWaiting();
});

async function handleFetch(request) {
  const url = new URL(request.url);
  if (request.method !== 'GET' || url.origin !== location.origin) return fetch(request);
  if (url.pathname.startsWith('/api/')) return networkFirst(request);   // 接口走网络优先
  if (/\.(css|js|png|woff2)$/.test(url.pathname)) return cacheFirst(request);
  return staleWhileRevalidate(request);
}
      

      

        三条硬性限制：① 只能在 HTTPS 或 localhost 下注册；
        ② file:// 下完全不可用（navigator.serviceWorker 都不存在）；
        ③ SW 脚本必须同源，作用域默认不能超出脚本所在目录。
